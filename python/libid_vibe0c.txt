# -*- coding: utf-8 -*-
"""
Randomized low‑rank linear‑algebra routines.

Implemented algorithms
----------------------
* ``orth_sketch`` – adaptive blocked randomized range finder.
* ``rrqr_randomized`` – rank‑revealing QR via a sketch.
* ``rrsvd_randomized`` – truncated SVD via a sketch.
* ``rrid_randomized`` – interpolative decomposition (two variants).

A small Hilbert‑matrix generator is also provided for testing.

The code works for **both real and complex** input matrices and strives
to be numerically stable while staying reasonably fast.

Author: Your Name
SPDX‑License‑Identifier: BSD‑3‑Clause
"""

from __future__ import annotations

from typing import Tuple, Optional

import numpy as np
from numpy.linalg import norm
from scipy import linalg

__all__ = [
    "orth_sketch",
    "rrqr_randomized",
    "rrsvd_randomized",
    "rrid_randomized",
    "hilbert_matrix",
]

# ----------------------------------------------------------------------
# Helper: random test matrices (Gaussian / Uniform)
# ----------------------------------------------------------------------


def _gaussian_omega(
    A: np.ndarray,
    n: int,
    block_size: int,
    *,
    rng: Optional[np.random.Generator] = None,
) -> np.ndarray:
    """
    Gaussian test matrix Ω with the same dtype as ``A``.

    Parameters
    ----------
    A : np.ndarray
        Reference matrix – its dtype determines the dtype of Ω.
    n, block_size : int
        Desired shape of Ω (``n`` rows, ``block_size`` columns).
    rng : np.random.Generator, optional
        Random number generator.  If omitted a fresh generator is created.

    Returns
    -------
    np.ndarray
        ``(n, block_size)`` Gaussian matrix.  For a complex ``A`` the
        real and imaginary parts are independent ``N(0, 1)`` draws.
    """
    if rng is None:
        rng = np.random.default_rng()

    if not np.iscomplexobj(A):
        return rng.standard_normal((n, block_size), dtype=A.dtype)

    # Complex case – draw real and imag parts separately.
    real = rng.standard_normal((n, block_size), dtype=A.real.dtype)
    imag = rng.standard_normal((n, block_size), dtype=A.real.dtype)
    return (real + 1j * imag).astype(A.dtype, copy=False)


def _uniform_omega(
    A: np.ndarray,
    n: int,
    block_size: int,
    *,
    rng: Optional[np.random.Generator] = None,
) -> np.ndarray:
    """
    Uniform test matrix Ω on ``[-1, 1)`` with the same dtype as ``A``.

    Parameters are the same as for :func:`_gaussian_omega`.
    """
    if rng is None:
        rng = np.random.default_rng()

    if not np.iscomplexobj(A):
        return rng.uniform(-1.0, 1.0, size=(n, block_size)).astype(A.dtype)

    real = rng.uniform(-1.0, 1.0, size=(n, block_size)).astype(A.real.dtype)
    imag = rng.uniform(-1.0, 1.0, size=(n, block_size)).astype(A.real.dtype)
    return (real + 1j * imag).astype(A.dtype, copy=False)


# ----------------------------------------------------------------------
# Power iteration (optional)
# ----------------------------------------------------------------------


def _power_iteration(
    A: np.ndarray,
    X: np.ndarray,
    n_iter: int,
) -> np.ndarray:
    """
    Apply ``n_iter`` subspace power iterations to ``X``.

    ``X`` has shape ``(n, k)``; after the loop it still has that shape but
    its columns are better aligned with the dominant left singular vectors of ``A``.
    """
    for _ in range(n_iter):
        # Orthogonalise the current sketch.
        Q, _ = linalg.qr(X, mode="economic")
        # Multiply by Aᵀ and orthogonalise again.
        Z = A.conj().T @ Q
        Qz, _ = linalg.qr(Z, mode="economic")
        # New sketch.
        X = A @ Qz
    return X


# ----------------------------------------------------------------------
# 1. Adaptive blocked randomized range finder
# ----------------------------------------------------------------------


def orth_sketch(
    A: np.ndarray,
    rtol: float,
    block_size: int = 42,
    flag_power: int = 0,
    *,
    rng: Optional[np.random.Generator] = None,
) -> Tuple[int, np.ndarray]:
    """
    Build an orthonormal basis for the column space of ``A`` using a
    blocked adaptive randomized range finder.

    The algorithm draws blocks of ``block_size`` random vectors, orthogonalises
    them against the basis already built, (optionally) applies a few power
    iterations, and stops as soon as the **relative Frobenius‑norm error**

    ``‖A – Q @ (Qᴴ A)‖_F / ‖A‖_F ≤ rtol``

    is satisfied.

    Parameters
    ----------
    A : (m, n) ndarray
        Input matrix (real or complex).
    rtol : float, > 0
        Desired relative error (Frobenius norm).
    block_size : int, default 42
        Initial number of random vectors per block (must be ≥ 1).
    flag_power : int, default 0
        Number of power‑iteration steps applied to each block.
    rng : np.random.Generator, optional
        Random generator for reproducibility.

    Returns
    -------
    k : int
        Number of basis vectors retained (``k ≤ min(m, n)``).
    Q : (m, k) ndarray
        Orthonormal basis for the (approximate) column space of ``A``.
        If ``k == 0`` an empty ``(m, 0)`` array is returned.
    """
    # -------------------------- sanity checks --------------------------
    if rtol <= 0:
        raise ValueError("rtol must be a positive number.")
    if block_size < 1:
        raise ValueError("block_size must be at least 1.")
    if flag_power < 0 or not isinstance(flag_power, int):
        raise ValueError("flag_power must be a non‑negative integer.")

    m, n = A.shape
    max_rank = min(m, n)

    if max_rank == 0:
        return 0, np.empty((m, 0), dtype=A.dtype)

    # -------------------------- initialisation -------------------------
    Q = np.empty((m, max_rank), dtype=A.dtype)   # pre‑allocate full size
    cols_written = 0
    A_norm = norm(A, ord="fro")
    A_res = A.copy()  # residual that we keep updating

    # -----------------------------------------------------------------
    # Main loop – we stop either because the tolerance is met or because
    # we have exhausted the full rank.
    # -----------------------------------------------------------------
    while cols_written < max_rank:
        # 1️⃣ draw a fresh test matrix
        Omega = _gaussian_omega(A, n, block_size, rng=rng)

        # 2️⃣ sketch the residual (power iteration optional)
        Y = A @ Omega
        if flag_power > 0:
            Y = _power_iteration(A, Y, flag_power)

        # 3️⃣ orthogonalise against the basis already built
        if cols_written > 0:
            Q_prev = Q[:, :cols_written]
            # Subtract the projection onto the existing space.
            Y -= Q_prev @ (Q_prev.conj().T @ Y)

        # 4️⃣ QR of the new block
        Qblk, _ = linalg.qr(Y, mode="economic")

        # 5️⃣ Update the residual
        Bblk = Qblk.conj().T @ A_res
        A_res -= Qblk @ Bblk

        # 6️⃣ Store the new orthonormal vectors (truncate if we would exceed max_rank)
        space_left = max_rank - cols_written
        to_store = min(space_left, Qblk.shape[1])
        Q[:, cols_written : cols_written + to_store] = Qblk[:, :to_store]
        cols_written += to_store

        # 7️⃣ Check stopping criterion
        rel_err = norm(A_res, ord="fro") / A_norm
        if rel_err <= rtol:
            break

        # 8️⃣ If we did not succeed, increase the block size (doubling works well)
        block_size = min(2 * block_size, max_rank - cols_written)
        if block_size == 0:               # nothing left to draw
            break

    # Trim to the actual size.
    Q = Q[:, :cols_written]
    k = Q.shape[1]
    return k, Q


# ----------------------------------------------------------------------
# 2. Rank‑revealing QR (randomised sketch)
# ----------------------------------------------------------------------


def _determine_rank_from_R(R: np.ndarray, A_norm: float, rtol: float) -> int:
    """
    Simple heuristic: count diagonal entries of ``R`` that are larger than
    ``rtol * A_norm`` (or ``rtol * max(diag(R))`` if ``A_norm`` is zero).
    """
    if R.size == 0:
        return 0
    diag = np.abs(np.diag(R))
    # Guard against pathological zero‑norm matrices.
    thresh = rtol * (A_norm if A_norm != 0 else diag[0])
    return int(np.sum(diag > thresh))


def rrqr_randomized(
    A: np.ndarray,
    rtol: float,
    block_size: int = 42,
    flag_power: int = 0,
    *,
    rng: Optional[np.random.Generator] = None,
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Rank‑revealing QR using a randomized sketch.

    Parameters
    ----------
    A : (m, n) ndarray
    rtol : float, > 0
        Relative tolerance used to decide the numerical rank.
    block_size, flag_power : as in :func:`orth_sketch`.
    rng : optional random generator.

    Returns
    -------
    Q : (m, k) orthonormal matrix
    R : (k, n) upper‑triangular matrix
    piv : permutation vector (length ``n``) such that ``A[:, piv] = Q @ R``.
    """
    m, n = A.shape
    A_norm = norm(A, ord="fro")
    k, Q_basis = orth_sketch(A, rtol, block_size, flag_power, rng=rng)

    # If the sketch already captured the full column space, just do a full QR.
    if k == min(m, n):
        Q, R, piv = linalg.qr(A, mode="economic", pivoting=True)
        rank = _determine_rank_from_R(R, A_norm, rtol)
        return Q[:, :rank], R[:rank, :], piv

    # Project A onto the basis we just built.
    A_proj = Q_basis.conj().T @ A
    Q_proj, R, piv = linalg.qr(A_proj, mode="economic", pivoting=True)
    Q = Q_basis @ Q_proj
    rank = _determine_rank_from_R(R, A_norm, rtol)
    return Q[:, :rank], R[:rank, :], piv


# ----------------------------------------------------------------------
# 3. Truncated SVD (randomised sketch)
# ----------------------------------------------------------------------


def rrsvd_randomized(
    A: np.ndarray,
    rtol: float,
    block_size: int = 42,
    flag_power: int = 0,
    *,
    rng: Optional[np.random.Generator] = None,
) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    """
    Truncated SVD via a randomized range finder.

    Returns the thin SVD of a low‑rank approximation of ``A``.
    """
    m, n = A.shape
    A_norm = norm(A, ord="fro")
    k, Q_basis = orth_sketch(A, rtol, block_size, flag_power, rng=rng)

    # Full rank – just call SciPy's SVD.
    if k == min(m, n):
        U, s, Vh = linalg.svd(A, full_matrices=False, lapack_driver="gesdd")
        rank = int(np.sum(s > rtol * s[0]))
        return U[:, :rank], s[:rank], Vh[:rank, :]

    # Project to the subspace spanned by Q_basis.
    B = Q_basis.conj().T @ A
    U_tilde, s, Vh = linalg.svd(B, full_matrices=False, lapack_driver="gesdd")
    U = Q_basis @ U_tilde
    rank = int(np.sum(s > rtol * s[0]))
    return U[:, :rank], s[:rank], Vh[:rank, :]


# ----------------------------------------------------------------------
# 4. Interpolative decomposition (randomised QR)
# ----------------------------------------------------------------------


def rrid_randomized(
    A: np.ndarray,
    rtol: float,
    block_size: int = 42,
    flag_power: int = 0,
    *,
    use_svd: bool = False,
    rng: Optional[np.random.Generator] = None,
) -> Tuple[int, np.ndarray, np.ndarray]:
    """
    Interpolative decomposition via a randomized QR.

    Parameters
    ----------
    A : (m, n) ndarray
    rtol : float, > 0
        Relative tolerance used for rank decisions *and* for the SVD‑based
        variant (if ``use_svd`` is ``True``).
    block_size, flag_power : as in :func:`orth_sketch`.
    use_svd : bool, optional
        If ``True`` solve the interpolation problem via a thin SVD of the
        leading block ``R11`` (more robust when ``R11`` is ill‑conditioned).
        If ``False`` (default) solve the triangular system directly.
    rng : optional random generator.

    Returns
    -------
    k : int
        Numerical rank (number of selected columns).
    piv : ndarray, shape (n,)
        Column permutation returned by the underlying QR.
    T : ndarray, shape (k, n‑k)
        Interpolation matrix such that ``A[:, piv[:k]] @ T ≈ A[:, piv[k:]]``.
    """
    # 1️⃣ Randomized rank‑revealing QR
    Q, R, piv = rrqr_randomized(A, rtol, block_size, flag_power, rng=rng)
    k = R.shape[0]

    # 2️⃣ Partition R = [R11 R12]
    R11 = R[:k, :k]      # (k, k) upper‑triangular
    R12 = R[:k, k:]      # (k, n‑k)

    if use_svd:
        # Compute a thin SVD of R11 and keep only the components
        # above the relative threshold.
        U, s, Vh = linalg.svd(R11, full_matrices=False)
        thresh = rtol * np.max(s)
        keep = s > thresh
        if not np.any(keep):
            # Everything is below the tolerance – return a zero matrix.
            T = np.zeros_like(R12)
        else:
            # Pseudoinverse of the retained part.
            inv_s = np.zeros_like(s)
            inv_s[keep] = 1.0 / s[keep]
            R11_pinv = (Vh.T @ np.diag(inv_s) @ U.T.conj())
            T = R11_pinv @ R12
    else:
        # Direct triangular solve (fast, but may be unstable for badly
        # conditioned R11 – that is exactly why the SVD option exists).
        T = linalg.solve_triangular(R11, R12, lower=False)

    return k, piv, T


# ----------------------------------------------------------------------
# 5. Hilbert matrix generator (public wrapper)
# ----------------------------------------------------------------------


def hilbert_matrix(m: int, n: int) -> np.ndarray:
    """
    Return the ``m × n`` Hilbert matrix.

    The (i, j) entry is ``1 / (i + j + 1)`` (1‑based indexing).

    Parameters
    ----------
    m, n : int
        Number of rows and columns.

    Returns
    -------
    np.ndarray, shape (m, n), dtype ``float64``.
    """
    if m < 0 or n < 0:
        raise ValueError("matrix dimensions must be non‑negative")
    i = np.arange(m)[:, None]
    j = np.arange(n)[None, :]
    return 1.0 / (i + j + 1.0)

        ||Q^* Q - I||_2 <= rtol,

    deliberately robust: an empty input matrix yields an empty basis and a
